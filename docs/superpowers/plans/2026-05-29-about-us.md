- [ ] **Step 2: Write the Playwright verification script**

Create `/tmp/verify-about.mjs`:

```js
import pw from '/Users/robodash/Desktop/Mosim/mosim-site/node_modules/playwright-core/index.js';
const { chromium } = pw;
import { mkdirSync } from 'fs';
const BASE = 'http://localhost:8123';
const SHOT = '/tmp/about-shots';
mkdirSync(SHOT, { recursive: true });

const results = [];
const check = (n, p, d = '') => { results.push({ n, p }); console.log(`${p ? 'PASS' : 'FAIL'}  ${n}${d ? ' — ' + d : ''}`); };

const browser = await chromium.launch();
const ctx = await browser.newContext({ viewport: { width: 1280, height: 1000 } });
const page = await ctx.newPage();
const errs = [];
page.on('console', m => { if (m.type() === 'error') errs.push(m.text()); });
page.on('pageerror', e => errs.push('PAGEERROR: ' + e.message));

await page.goto(BASE + '/index.html', { waitUntil: 'networkidle' });

// Section exists
const about = page.locator('#about');
check('#about section exists', (await about.count()) > 0);

// Photo loaded (naturalWidth > 0)
const imgOk = await page.evaluate(() => {
  const img = document.querySelector('#about .about-photo img');
  return img && img.complete && img.naturalWidth > 0;
});
check('Founder photo loads (naturalWidth>0)', imgOk);

// Key copy present
const txt = (await about.textContent()) || '';
check('Headline present', txt.includes('I built the trip I wish my own family'));
check('Mosim word present', txt.includes('모심'));
check('Signature present', txt.includes('Sunggun Michael Jang'));

// Placeholder signature gone from page
const body = (await page.locator('body').textContent()) || '';
check('Old placeholder signature removed', !body.includes('Soo') || !body.includes('Head Concierge'));

// Desktop screenshot of the section
await about.scrollIntoViewIfNeeded();
await page.waitForTimeout(400);
await about.screenshot({ path: SHOT + '/about-desktop.png' });

// Nav anchor resolves: click About Us, confirm #about is in view
await page.goto(BASE + '/index.html', { waitUntil: 'networkidle' });
await page.click('nav.top a[href="#about"]');
await page.waitForTimeout(800);
const inView = await page.evaluate(() => {
  const el = document.getElementById('about');
  const r = el.getBoundingClientRect();
  return r.top < window.innerHeight && r.bottom > 0;
});
check('Nav "About Us" scrolls section into view', inView);

// Mobile layout (stacked)
const mobile = await browser.newContext({ viewport: { width: 390, height: 844 } });
const mp = await mobile.newPage();
await mp.goto(BASE + '/index.html', { waitUntil: 'networkidle' });
const mAbout = mp.locator('#about');
await mAbout.scrollIntoViewIfNeeded();
await mp.waitForTimeout(400);
await mAbout.screenshot({ path: SHOT + '/about-mobile.png' });
// On mobile the grid should be single-column: photo width ≈ container width region
const stacked = await mp.evaluate(() => {
  const grid = document.querySelector('#about .about-grid');
  const cols = getComputedStyle(grid).gridTemplateColumns.split(' ').length;
  return cols === 1;
});
check('Mobile: about-grid is single column', stacked);

console.log('\n=== console errors ===');
console.log(errs.length ? [...new Set(errs)].join('\n') : '(none)');
const passed = results.filter(r => r.p).length;
console.log(`\n=== ${passed}/${results.length} passed ===`);
await browser.close();
process.exit(passed === results.length ? 0 : 1);
```
